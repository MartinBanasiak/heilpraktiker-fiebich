    [EventSubscriber(Page,5447477,OnBeforeFinishAction,"",Skip,Skip)]
    LOCAL PROCEDURE OnBeforeFinishAction@1000000034(CouponLine@1000000000 : Record 5447472;VAR SalesHeader@1000000001 : Record 36;VAR SalesLine@1000000002 : Record 37;CouponDiscountAmount@1000000003 : Decimal;VAR IsHandled@1000000004 : Boolean);
    VAR
      SalesLineAmount@1000000005 : Decimal;
      ShippingOptionsManagement@1000000006 : Codeunit 5447450;
      ShippingOption@1000000007 : Record 5447453;
      WebshopShippingOptionLink@1000000008 : Record 5447311;
      TempShippingOption@1000000009 : TEMPORARY Record 5447453;
      WebshopCouponShippingLine@1000000010 : Record 5447330;
      NewShippingCost@1000000011 : Decimal;
    BEGIN
      IF NOT IsHandled THEN BEGIN
        IF CouponLine."Value Type Shop" = CouponLine."Value Type Shop"::Sonderversand THEN BEGIN
          SalesLineAmount := ShippingOptionsManagement.SetShippingOptionsForSalesHeader(ShippingOption,SalesHeader);
          IF ShippingOption.FINDSET THEN REPEAT
            TempShippingOption.INIT;
            TempShippingOption.TRANSFERFIELDS(ShippingOption);
            TempShippingOption.INSERT;

            IF WebshopCouponShippingLine.GET(CouponLine.Code,CouponLine."Coupon Code") THEN BEGIN
              IF (WebshopCouponShippingLine."Shipping Coupon Type" = WebshopCouponShippingLine."Shipping Coupon Type"::amount) AND (WebshopCouponShippingLine."Shipping Coupon Value" > 0) THEN BEGIN
                TempShippingOption."Shipping Cost" := WebshopCouponShippingLine."Shipping Coupon Value";
                TempShippingOption.MODIFY;
              END;
              IF (WebshopCouponShippingLine."Shipping Coupon Type" = WebshopCouponShippingLine."Shipping Coupon Type"::percent) AND (WebshopCouponShippingLine."Shipping Coupon Value" > 0) THEN BEGIN
                NewShippingCost := TempShippingOption."Shipping Cost" - (TempShippingOption."Shipping Cost" / 100 * WebshopCouponShippingLine."Shipping Coupon Value");
                TempShippingOption."Shipping Cost" := NewShippingCost;
                TempShippingOption.MODIFY;
              END;
            END;
          UNTIL ShippingOption.NEXT = 0;
          IF ShippingOptionsManagement.SelectShippingOptionFromShippingOptions(TempShippingOption,SalesLineAmount) THEN BEGIN
            ShippingOptionsManagement.AddShippingOptionLineToSalesHeader(SalesHeader,TempShippingOption,SalesLine,SalesLineAmount);
          END;
          IsHandled := TRUE;
        END;
      END;
    END;
