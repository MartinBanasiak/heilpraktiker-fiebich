    VAR

      Error001@1000000005 : TextConst 'DEU=Ein Datensatz f�r eine Marketplace-Kategorie konnte nicht synchronisiert werden.;ENU=A Record for a Marketplace-Category could not be synchronized.';

        AddVar(GetRec,'filter[parent_category_code]','');

        IF CountRecords THEN BEGIN
          WebshopManagement.StateIndicatorRegisterTable(Text001,NoOfRec)
        END ELSE BEGIN
          //Verschiebe nachfolgende Kategorien
          LineNoInterval := 10000;
          MoveAllCategoriesBelowLineNo(ShopCode,ShopLanguageCode,AboveLineNo,(NoOfRec * LineNoInterval));

          WebshopManagement.StateIndicatorNewTable(Text001,NoOfRec);
          i := 1;
          WHILE (i <= NoOfRec) DO BEGIN
            CategoryFetched := GetNextFetchedMktplCategory(ShopCode,ShopLanguageCode,AboveLineNo);

              i += 1;
              WebshopManagement.StateIndicatorUpdate;
            END ELSE BEGIN

            END;
          END;
        END;
      END;

    PROCEDURE GetNextFetchedMktplCategory@1000000208(ShopCode@1000000013 : Code[30];ShopLanguageCode@1000000014 : Code[10];AboveLineNo@1000000018 : Integer) Success : Boolean;
    VAR
      Delimiter@1000000001 : Text[5];
      SendRec@1000000002 : Text[1024];
      GetRec@1000000007 : Text[1024];
      GetResponseText@1000000005 : Text[1024];
      Category@1000000000 : Record 5447303;
      LogFile@1000000004 : File;
      AboveCategoryCode@1000000003 : Text[40];
      RecID@1000000006 : Integer;
      RecMarketplaceType@1000000008 : Option;
      RecCategoryCode@1000000009 : Text[45];
      RecName@1000000012 : Text[45];
      RecLevel@1000000010 : Integer;
      RecParentCode@1000000011 : Text[45];
      RecRootCode@1000000019 : Text[45];
      CategoryInsert@1000000015 : Boolean;
      SetRec@1000000016 : Text[1024];
      NewLineNo@1000000017 : Integer;
    BEGIN
      WITH Category DO BEGIN
        Delimiter := '-;-';
        CLEAR(GetRec);
        CLEAR(SendRec);
        AddVar(SendRec,'table','shop_marketplace_categories');
        AddVar(SendRec,'delimiter',Delimiter);
        AddVar(SendRec,'fields','id,marketplace_type,category_code,category_name,' +
               'category_level,parent_category_code,root_category_code');
        IF AboveLineNo > 0 THEN BEGIN
          Category.RESET;
          Category.GET(ShopCode,ShopLanguageCode,AboveLineNo);
          AddVar(SendRec,'filter[parent_category_code]',Category."Search Code");
        END ELSE BEGIN
          AddVar(SendRec,'filter[parent_category_code]','');
        END;

        AddVar(SendRec,'filter[update_insert]',TRUE);
        GetRec := WebshopManagement.GetRequest('get_content',SendRec);
        IF STRPOS(GetRec,Delimiter) <> 0 THEN BEGIN
          EVALUATE(RecID,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecMarketplaceType,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecCategoryCode,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecName,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecLevel,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecParentCode,WebshopManagement.Explode(GetRec,Delimiter));
          EVALUATE(RecRootCode,WebshopManagement.Explode(GetRec,Delimiter));

          AboveCategoryCode := Category.GetAboveCodeForParentCode(ShopCode,ShopLanguageCode,RecParentCode);
          Category.RESET;
          NewLineNo := Category.GetNextCategoryLineNo(ShopCode,ShopLanguageCode,AboveCategoryCode);
          Category.INIT;
          VALIDATE(Category."Shop Code",ShopCode);
          VALIDATE(Category."Language Code",ShopLanguageCode);
          VALIDATE(Category.Name,RecName);
          VALIDATE(Category.Level,RecLevel);
          VALIDATE(Category."Search Code",RecCategoryCode);
          VALIDATE(Category."Line No.",NewLineNo);
          IF RecRootCode <> '' THEN
            VALIDATE(Category."Marketplace Root Category Code",RecRootCode)
          ELSE
            VALIDATE(Category."Marketplace Root Category Code",RecCategoryCode);
          CategoryInsert := Category.INSERT;
          IF CategoryInsert THEN BEGIN
              CLEAR(SetRec);
              AddVar(SetRec,'table','shop_marketplace_categories');
              AddVar(SetRec,'primary[id]',RecID);
              AddVar(SetRec,'data[update_insert]',FALSE);
              WebshopManagement.SendRequest('update',SetRec);

              //FetchCategoryAttributes(Category,TRUE);

          END;
          COMMIT;
          EXIT(CategoryInsert);
        END ELSE BEGIN
          IF GetRec<>'0' THEN BEGIN
            IF NOT WebshopSetup."Catch & Log WebRequest Errors" THEN
              ERROR(GetRec)
