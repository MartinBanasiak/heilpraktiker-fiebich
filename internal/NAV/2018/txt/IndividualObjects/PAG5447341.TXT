    PROCEDURE SetTypeIsOrderOption@1000000000();
    BEGIN
      IF Type = Type::SubscriptionOrderOption THEN BEGIN
        TypeIsSequenceItem := FALSE;
        TypeIsOrderOption := TRUE;
        "Payment Type" := "Payment Type"::PerTurn;
      END ELSE BEGIN
        TypeIsSequenceItem := TRUE;
        TypeIsOrderOption := FALSE;
      END;
    END;

    PROCEDURE SetTypeIsSequenceItem@1000000001();
    BEGIN
      IF Type = Type::SubscriptionSequenceItem THEN BEGIN
        TypeIsOrderOption := FALSE;
        TypeIsSequenceItem := TRUE;
      END ELSE BEGIN
        "Payment Type" := "Payment Type"::PerTurn;
        TypeIsOrderOption := TRUE;
        TypeIsSequenceItem := FALSE;
      END;
    END;

    LOCAL PROCEDURE SetCanSetLateEntryPossible@1000000003();
    BEGIN
      IF (Type = Type::SubscriptionSequenceItem)
         AND ("Start Type" = "Start Type"::Join)
         AND ("Payment Type" = "Payment Type"::Once)  THEN BEGIN
        CanSetLateEntryPossible := TRUE;
      END ELSE BEGIN
        CanSetLateEntryPossible := FALSE;
        IF "Allow Late Entry" = TRUE THEN BEGIN
          "Allow Late Entry" := FALSE;
          MODIFY(FALSE);
        END;
      END;
    END;
