    PROCEDURE EditContent@1000000002();
    BEGIN
      IF ClientAddInType = ClientAddInType::Undefined THEN
        ERROR(Text007);
      IF Type = Type::YoutubeID THEN
        EXIT;
      WebshopUpdateManagement.SetSendNow(TRUE);
      WebshopUpdateManagement.UpdateWebshopItemFile(Rec);
      GET("Shop Code","Language Code","Item No.",Type,"Variant Code","Line No.");
      COMMIT;

      IF ClientAddInType = ClientAddInType::DynamicCommerce THEN BEGIN
        //DynamicCommerceAddIn
        CLEAR(WebshopWebform);
        WebshopWebform.SetWebformType(WebformType::ItemFile);
        WebshopWebform.SetShopCode("Shop Code");
        WebshopWebform.SetLanguageCode("Language Code");
        WebshopWebform.SetItemNo("Item No.");
        WebshopWebform.SetItemFileType(Type);
        WebshopWebform.SetLineNo("Line No.");
        WebshopWebform.LOOKUPMODE(TRUE);
        WebshopWebform.RUNMODAL();
      END ELSE BEGIN
        //Legacy
        CLEAR(EditItemFile);
        TempItemFile.SETRANGE(Type,Type);
        TempItemFile.SETRANGE("Shop Code","Shop Code");
        TempItemFile.SETRANGE("Language Code","Language Code");
        TempItemFile.SETRANGE("Item No.","Item No.");
        TempItemFile.SETRANGE("Variant Code","Variant Code");
        TempItemFile.SETRANGE("Line No.","Line No.");
        EditItemFile.SETTABLEVIEW(TempItemFile);
        EditItemFile.LOOKUPMODE(TRUE);
        EditItemFile.RUNMODAL();
      END;

      GET("Shop Code","Language Code","Item No.",Type,"Variant Code","Line No.");
      WebshopUpdateManagement.CheckWebshopItemFile(Rec);
      GET("Shop Code","Language Code","Item No.",Type,"Variant Code","Line No.");
      WebshopUpdateManagement.SetSendNow(FALSE);
      IF File THEN
        FileText := YesText
      ELSE
        FileText := NoText;

      CurrPage.UPDATE;
    END;
