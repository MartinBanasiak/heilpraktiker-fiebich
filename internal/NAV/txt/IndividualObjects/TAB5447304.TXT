    PROCEDURE WebshopInventory@1000000000(VariantCode@1000000001 : Code[10]) WebshopInventory : Decimal;
    VAR
      Item@1000000000 : Record 27;
    BEGIN
      // Lagerbestand des Artikels berechnen
      Shop.GET("Shop Code");
      Item.GET("Item No.");

      // Abh�ngig von Parameter auf Variante filtern
      IF VariantCode <> '' THEN
        Item.SETRANGE("Variant Filter",VariantCode);

      // Abh�ngig von Shop-Einrichtung Lagerortfilter ber�cksichtigen
      IF Shop."Location Code Filter" <> '' THEN
        Item.SETFILTER("Location Filter",Shop."Location Code Filter");

      // Lagerbestand aus Artikel berechnen
      Item.CALCFIELDS(Inventory);
      WebshopInventory := Item.Inventory;

      // Abh�ngig von Shop-Einrichtung Menge in Verkaufsauftrag abziehen
      IF Shop."Inv. subtr. Sales Ord. (Days)" <> 0 THEN BEGIN
        Item.SETFILTER("Date Filter",'..' + FORMAT(CALCDATE('<CD+' + FORMAT(Shop."Inv. subtr. Sales Ord. (Days)") + 'D>')));
        Item.CALCFIELDS("Qty. on Sales Order");
        WebshopInventory -= Item."Qty. on Sales Order";
      END;

      // Abh�ngig von Shop-Einrichtung  Menge in Einkaufsbestellung addieren
      IF Shop."Inv. add Purch Ord. (Days)" <> 0 THEN BEGIN
        Item.SETFILTER("Date Filter",'..' + FORMAT(CALCDATE('<CD+' + FORMAT(Shop."Inv. add Purch Ord. (Days)") + 'D>')));
        Item.CALCFIELDS("Qty. on Purch. Order");
        WebshopInventory += Item."Qty. on Purch. Order";
      END;

      IF WebshopInventory < 0 THEN
        WebshopInventory := 0;
    END;

    PROCEDURE WebshopRetailPrice@1000000001() RetailPrice : Decimal;
    VAR
      Item@1000000006 : Record 27;
      WebshopItem@1000000005 : Record 5447304;
      "Webshop Item Link"@1000000004 : Record 5447308;
      SalesPrice@1000000003 : Record 7002;
      VATPostingSetup@1000000007 : Record 325;
      AllowLineDisc@1000000002 : Boolean;
      AllCustPrice@1000000001 : Decimal;
      GroupPrice@1000000000 : Decimal;
    BEGIN
      // Ermittelt Webshop-Listenpreis anhand Shop-Einrichtung

      WITH Item DO BEGIN
        Shop.GET("Shop Code");
        IF GET("Item No.") THEN BEGIN
          WebshopItemLink.SETRANGE(Type,WebshopItemLink.Type::Variant);
          WebshopItemLink.SETRANGE("Shop Code",Shop.Code);
          WebshopItemLink.SETRANGE("Item No.","No.");
          IF (Shop."Variant Type" = Shop."Variant Type"::ShopVariantsParentItem)
          AND (WebshopItemLink.FINDSET) THEN BEGIN
            REPEAT
              WebshopItem.SETRANGE("Shop Code",Shop.Code);
              WebshopItem.SETRANGE("Language Code","Language Code");
              WebshopItem.SETRANGE("Item No.",WebshopItemLink."Linked Item No.");
              WebshopItem.SETRANGE("Is Active",TRUE);
              WebshopItem.SETFILTER(WebshopItem."Starting Date",'%1 | ..%2', 0D, TODAY);
              WebshopItem.SETFILTER(WebshopItem."Ending Date",'%1 | %2..', 0D, TODAY);
              IF WebshopItem.FINDFIRST THEN
                IF (RetailPrice > WebshopItem.WebshopBasePrice) OR (RetailPrice = 0) THEN
                  RetailPrice := WebshopItem.WebshopRetailPrice;
            UNTIL WebshopItemLink.NEXT = 0;
          END ELSE BEGIN
            CASE Shop."Retail Price Type" OF
              Shop."Retail Price Type"::"Sales Price": BEGIN
                RetailPrice := Item."Unit Price";
                IF Shop."Display Prices Incl. VAT" AND NOT "Price Includes VAT" THEN
                  IF VATPostingSetup.GET(Shop."VAT Bus. Posting Group","VAT Prod. Posting Group") THEN
                    RetailPrice := RetailPrice / 100 * (100 + VATPostingSetup."VAT %");
                IF NOT Shop."Display Prices Incl. VAT" AND "Price Includes VAT" THEN
                  IF VATPostingSetup.GET(Shop."VAT Bus. Posting Group","VAT Prod. Posting Group") THEN
                    RetailPrice := RetailPrice / (100 + VATPostingSetup."VAT %") * 100;
              END;
              Shop."Retail Price Type"::"Customer Price Group": BEGIN
                AllCustPrice := WebshopGetPrice(TRUE,'','',Shop."Display Prices Incl. VAT",Shop."VAT Bus. Posting Group");
                RetailPrice := WebshopGetPrice(FALSE,Shop."Cust. Price Group Retail Price",Shop."Cust. Disc. Group Retail Price",
                  Shop."Display Prices Incl. VAT",Shop."VAT Bus. Posting Group");
                IF ((RetailPrice > AllCustPrice) AND (AllCustPrice <> 0)) OR (RetailPrice = 0) THEN
                  RetailPrice := AllCustPrice;
              END;
              Shop."Retail Price Type"::"All Customer Price": RetailPrice := WebshopGetPrice(TRUE,'','',
                Shop."Display Prices Incl. VAT",Shop."VAT Bus. Posting Group");
            END;
          END;
        END;
      END;
    END;

    PROCEDURE WebshopGetPrice@1000000004(AllCustomers@1000000000 : Boolean;CustPriceGroup@1000000001 : Code[10];CustDiscGroup@1000000002 : Code[10];IncVAT@1000000012 : Boolean;VATBusPostingGroup@1000000014 : Code[10]) BestPrice : Decimal;

            BestPrice := SalesPrice."Unit Price";
            AllowLineDisc := SalesPrice."Allow Line Disc.";
            BestPriceIncVAT := SalesPrice."Price Includes VAT";

      END ELSE BEGIN
        IF AllCustomers THEN BEGIN

          BestPriceIncVAT := Item."Price Includes VAT";
        END;
      END;

      IF (BestPrice <> 0) AND (AllowLineDisc) THEN BEGIN
        BestLineDiscount := WebshopLineDiscount(TRUE,FALSE,'');

        IF TempLineDiscount > BestLineDiscount THEN
          BestLineDiscount := TempLineDiscount;

            BestLineDiscount := TempLineDiscount;

          IF TempLineDiscount > BestLineDiscount THEN
            BestLineDiscount := TempLineDiscount;
        END;
        IF BestLineDiscount <> 0 THEN BEGIN
          BestPrice := BestPrice / 100 * (100-BestLineDiscount);
        END;
      END;

      IF IncVAT AND NOT BestPriceIncVAT THEN
        IF VATPostingSetup.GET(VATBusPostingGroup,Item."VAT Prod. Posting Group") THEN

          BestPrice := BestPrice / (100 + VATPostingSetup."VAT %") * 100;
