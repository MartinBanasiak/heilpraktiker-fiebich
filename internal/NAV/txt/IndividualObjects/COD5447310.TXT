    PROCEDURE UpdateWebshopOrderSalesChart@1000000000(VAR BusinessChartBuffer@1000000000 : Record 485;MovePeriod@1000000001 : ' ,Next,Previous';PeriodLength@1000000002 : 'Day,Week,Month,Quarter,Year');
    VAR
      WebshopShop@1000000003 : Record 5447301;
      Calendar@1000000004 : Record 2000000007;
      XIndex@1000000005 : Integer;
      YIndex@1000000012 : Integer;
      CalendarEntry@1000000006 : Record 99000757;
      Amount@1000000007 : ARRAY [200] OF Decimal;
      "Amount (incl. VAT)"@1000000008 : ARRAY [200] OF Decimal;
      WebshopCode@1000000009 : ARRAY [200] OF Code[20];
      ShopCounter@1000000010 : Integer;
      ShopCodeFilterString@1000000011 : Text;
      SalesInvoiceHeader@1000000013 : Record 112;
      SalesCrMemoHeader@1000000014 : Record 114;
      CurrColumnIndex@1000000017 : Integer;
      NoOfColumns@1000000018 : Integer;
      LocAmnt@1000000015 : Decimal;
      LocAmntVAT@1000000016 : Decimal;
      InvoiceCount@1000000019 : Integer;
      CrMemoCount@1000000020 : Integer;
    BEGIN
      WITH BusinessChartBuffer DO BEGIN
        Initialize;

      //  gWorkdate := GetEndDateForCurrInterval(gWorkdate,PeriodLength);

        //Generate the period to be displayed
        SetChartPeriod(PeriodLength, MovePeriod,gWorkdate);

        //Add Measures for Webshop
        ShopCounter := 0;
        ShopCodeFilterString := '';
        IF WebshopShop.FINDSET THEN REPEAT
          AddMeasure(WebshopShop.Code,1,"Data Type"::Decimal, gChartType);
          gShopCodeArray[ShopCounter + 1] := WebshopShop.Code;
          IF ShopCounter > 0 THEN
            ShopCodeFilterString += '|';
          ShopCodeFilterString += WebshopShop.Code;
          Amount[ShopCounter + 1] := 0;
          "Amount (incl. VAT)"[ShopCounter + 1] := 0;
          ShopCounter += 1;
        UNTIL WebshopShop.NEXT = 0;

        //Set X-Axis  Value
        SetXAxis(Text004, "Data Type"::String);

        XIndex := 0;
        CurrStartDate := GetStartDateForPeriodLength(PeriodLength);
        CurrEndDate := GetEndDateForCurrInterval(CurrStartDate,PeriodLength);
        NoOfColumns := GetNoOfColumnsForPeriod(PeriodLength);
        gStartDate := GetStartDateForPeriodLength(PeriodLength);
        gEndDate :=  GetEndDateForPeriodLength(PeriodLength);
        FOR CurrColumnIndex := 1 TO NoOfColumns DO BEGIN

          gDateFilterArray[CurrColumnIndex] := FORMAT(CurrStartDate,0,'<Standard Format,5>')+'..'+FORMAT(CurrEndDate,0,'<Standard Format,5>');

          AddColumn(GetCurrPeriodName(CurrStartDate,PeriodLength));

          //Calculate Totals for All Webshops
          FOR YIndex := 0 TO ShopCounter - 1 DO BEGIN
            LocAmnt := 0;
            LocAmntVAT := 0;
            //Add Posted Sales Invoices
            SalesInvoiceHeader.RESET;
            SalesInvoiceHeader.SETRANGE("Posting Date",CurrStartDate,CurrEndDate);
            SalesInvoiceHeader.SETRANGE("Webshop Shop Code",gShopCodeArray[YIndex + 1]);
            InvoiceCount := 0;
            IF SalesInvoiceHeader.FINDSET THEN REPEAT
              InvoiceCount += 1;
              SalesInvoiceHeader.CALCFIELDS(Amount,"Amount Including VAT");
              LocAmnt += SalesInvoiceHeader.Amount;
              LocAmntVAT += SalesInvoiceHeader."Amount Including VAT";
            UNTIL SalesInvoiceHeader.NEXT = 0;

            //Subtract Posted Sales Cr Memos
            SalesCrMemoHeader.RESET;
            SalesCrMemoHeader.SETRANGE("Posting Date",CurrStartDate,CurrEndDate);
            SalesCrMemoHeader.SETRANGE("Webshop Shop Code",gShopCodeArray[YIndex + 1]);
            CrMemoCount := 0;
            IF SalesCrMemoHeader.FINDSET THEN REPEAT
              CrMemoCount += 1;
              SalesCrMemoHeader.CALCFIELDS(Amount,"Amount Including VAT");
              LocAmnt -= SalesCrMemoHeader.Amount;
              LocAmntVAT -= SalesCrMemoHeader."Amount Including VAT";
            UNTIL SalesCrMemoHeader.NEXT = 0;

            // Set the values for each of the stacks
            CASE gMeasure OF
              gMeasure::Amount:
                SetValueByIndex(YIndex,XIndex,LocAmnt);
              gMeasure::AmountVAT:
                SetValueByIndex(YIndex,XIndex,LocAmntVAT);
              gMeasure::NoOfPostedInvoices:
                SetValueByIndex(YIndex,XIndex,InvoiceCount);
              gMeasure::NoOfPostedCrMemos:
                SetValueByIndex(YIndex,XIndex,CrMemoCount);
            END;
          END;

          CurrEndDate := GetEndDateForNextInterval(CurrStartDate,PeriodLength);
          CurrStartDate := GetStartDateForNextInterval(CurrStartDate,PeriodLength);

          XIndex += 1;

        END;


      END;
    END;
