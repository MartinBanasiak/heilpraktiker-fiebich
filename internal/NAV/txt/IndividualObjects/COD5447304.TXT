    PROCEDURE UpdateDynamicData@1000000049(ShopCodeFilter@1000000000 : Text);
    BEGIN
      WebshopSetup.GET;

      IF WebshopSetup."Server Synch. Endpoint URL" <> '' THEN BEGIN

        WebshopUpdateBuffer.SendRequest('','',TRUE,FALSE);

        // Kundenspezifische Aktualisierung Bewegungsdaten
        CustWebshopUpdateManagement.StateIndicatorOpen(Text002);
        CountRecords := TRUE;
        FOR i := 1 TO 2 DO BEGIN
          CustUpdate();
          CountRecords := FALSE;
        END;
        CustWebshopUpdateManagement.StateIndicatorClose();


        // Allgemeine Aktualisierung Bewegungsdaten
        WebshopUpdateManagement.StateIndicatorOpen(Text001);
        CountRecords := TRUE;
        FOR i := 1 TO 2 DO BEGIN
          UpdateWithShopFilter(ShopCodeFilter);
          CountRecords := FALSE;
        END;
        WebshopUpdateManagement.StateIndicatorClose();


        // Marketplace Aktualisierung Bewegungsdaten
        IF WebshopSetup."Update Marketplaces" THEN BEGIN
          WebshopMarketplaceUpdate.StateIndicatorOpen(Text001);
          CountRecords := TRUE;
          FOR i := 1 TO 2 DO BEGIN
            MarketplaceUpdate();
            CountRecords := FALSE;
          END;
          WebshopMarketplaceUpdate.StateIndicatorClose();
        END;


        // Datum Aktualisierung Bewegungsdaten in Webshop-Setup setzten
        WebshopSetup.GET;
        WebshopSetup."Time Last Update Dynamic Data" := CURRENTDATETIME;
        IF NOT GUIALLOWED THEN
          WebshopSetup."Time Last Auto Upd. Dyn. Data" := CURRENTDATETIME;
        WebshopSetup.MODIFY;

        WebshopUpdateBuffer.SendRequest('','',TRUE,FALSE);

      END ELSE BEGIN
        IF GUIALLOWED THEN
          ERROR(Text003);
      END;
    END;

    LOCAL PROCEDURE UpdateWithShopFilter@1000000004(ShopCodeFilter@1000000000 : Text);
    VAR
      WebshopShop@1000000001 : Record 5447301;
    BEGIN
      IF ShopCodeFilter <> '' THEN BEGIN
        WebshopShop.SETFILTER(Code,ShopCodeFilter);
        IF WebshopShop.FINDSET(FALSE,FALSE) THEN REPEAT
          IF WebshopSetup."Fetch Orders" THEN
            WebshopUpdateManagement.GetAllSalesHeader(CountRecords,WebshopShop.Code);
        UNTIL WebshopShop.NEXT = 0;
      END ELSE BEGIN
        Update();
      END;
    END;

    PROCEDURE Update@1000000002();
    BEGIN
      // Aktualisierung Bewegungsdaten

      IF GUIALLOWED THEN BEGIN
        IF WebshopSetup."Fetch Orders" THEN
            WebshopUpdateManagement.GetAllSalesHeader(CountRecords,'');
      END;

      IF WebshopSetup."Fetch Return Orders" THEN
        WebshopUpdateManagement.GetAllReturnOrder(CountRecords);
      WebshopUpdateManagement.ProcessAllInvoiceOnlinePayment(CountRecords);
      WebshopUpdateManagement.ProcessAllCrMemoOnlinePayment(CountRecords);
      WebshopUpdateManagement.ClearAllInvoiceOnlinePayment(CountRecords);
      WebshopUpdateManagement.ClearAllCrMemoOnlinePayment(CountRecords);
      IF WebshopSetup."Update Sales Orders" THEN
        WebshopUpdateManagement.UpdateAllSalesOrder(CountRecords);
      IF WebshopSetup."Update Item Inventory" THEN
        WebshopUpdateManagement.UpdateAllWebshopItemInventory(CountRecords);
      IF WebshopSetup."Update Sales Prices" THEN
        WebshopUpdateManagement.UpdateAllWebshopItemStdPrices(CountRecords);
      IF WebshopSetup."Update Item Inventory" AND WebshopSetup."Update Item Variants" THEN
        WebshopUpdateManagement.UpdateAllItemVariantInventory(CountRecords);
      IF WebshopSetup."Update Posted Sales Invoices" THEN
        WebshopUpdateManagement.GetAllInvoiceRequest(CountRecords);
      IF WebshopSetup."Update Posted Sales Cr. Memos" THEN
        WebshopUpdateManagement.GetAllCrMemoRequest(CountRecords);
      IF WebshopSetup."Update Posted Sales Shipments" THEN
        WebshopUpdateManagement.GetAllShipmentRequest(CountRecords);
      IF WebshopSetup."Update Item Reviews" THEN BEGIN
        WebshopUpdateManagement.GetAllItemComments(CountRecords);
        WebshopUpdateManagement.UpdateAllWebshopItemComment(CountRecords);
      END;

      WebshopUpdateManagement.UpdateAllReturnReason(CountRecords);

      IF WebshopSetup."Create Subscription Orders" THEN
        WebshopUpdateManagement.ManageSubscriptions(CountRecords);

      IF NOT GUIALLOWED THEN BEGIN
        WebshopUpdateManagement.SendAllOrderConfirmMail(CountRecords);
        WebshopUpdateManagement.SendAllOrderShippingMail(CountRecords);
        WebshopUpdateManagement.SendAllOrderInvoiceMail(CountRecords);
      END;

      IF (WebshopSetup."Update Posted Sales Shipments") AND (WebshopSetup."Confirm Marketplace Orders") THEN
        WebshopUpdateManagement.UpdateAllSalesShipmentHeader(CountRecords);

      IF WebshopSetup."Send Orders to Newsl. System" THEN
        WebshopUpdateManagement.TransferNewsletter(CountRecords);
      IF NOT CountRecords THEN
        WebshopUpdateManagement.GetAndSaveLastSolrUpdateDT();
    END;

    PROCEDURE CustUpdate@1000000003();
    BEGIN
      // Kundenspezifische Aktualisierung Bewegungsdaten
    END;

    PROCEDURE MarketplaceUpdate@1000000001();
    BEGIN
      // Marketplace aktualisierung
      WebshopSetup.GET;
      IF WebshopSetup."Update Marketplaces" THEN BEGIN
        IF WebshopSetup."Update Marketplace Prices" THEN
          WebshopMarketplaceUpdate.UpdateMarketplacePrice(CountRecords);
        IF WebshopSetup."Update Marketplace Inventory" THEN
          WebshopMarketplaceUpdate.UpdateMarketplaceInventory(CountRecords);
        IF WebshopSetup."Fetch Marketplace Orders" THEN
          WebshopMarketplaceUpdate.UpdateMarketplaceOrders(CountRecords);
        IF WebshopSetup."Confirm Marketplace Orders" THEN
          WebshopMarketplaceUpdate.UpdateMarketplaceOrderState(CountRecords);
        IF WebshopSetup."Fetch Marketplace Upd. Errors" THEN BEGIN
          WebshopMarketplaceUpdate.GetAllMarketplaceErrors(CountRecords);
          WebshopMarketplaceUpdate.GetAllMarketplaceItemErrors(CountRecords);
        END;
      END;
    END;

    PROCEDURE ResetJobQueue@1000000000();
    BEGIN
      JobQueueEntry.SETFILTER("Object ID to Run",'5447304');
      JobQueueEntry.SETFILTER(Status,'%1',JobQueueEntry.Status::Error);
      IF JobQueueEntry.FINDFIRST THEN BEGIN
        JobQueueEntry.SetStatus(JobQueueEntry.Status::Ready);
        JobQueueEntry.MODIFY(TRUE);
        COMMIT;
      END;
    END;
