    PROCEDURE SetWebshopFilter@1000000002();
    VAR
      ItemNo@1000000000 : Code[20];
    BEGIN
      // Shop ermitteln
      IF NOT Shop.GET(ShopFilter) THEN BEGIN
        ShopFilter := '';
        IF GETFILTER("Shop Code") <> '' THEN
          IF Shop.GET(GETFILTER("Shop Code")) THEN
            ShopFilter := Shop.Code;
        IF (ShopFilter = '') AND (WebshopSetup."Default Shop Code" <> '') THEN
          IF Shop.GET(WebshopSetup."Default Shop Code") THEN
            ShopFilter := Shop.Code;
        IF ShopFilter = '' THEN
          IF Shop.FINDFIRST THEN
            ShopFilter := Shop.Code;
      END;

      // Sprache ermitteln
      IF ShopFilter <> '' THEN BEGIN
        IF NOT Language.GET(ShopFilter,LanguageFilter) THEN BEGIN
          LanguageFilter := '';
          Language.SETRANGE("Shop Code",ShopFilter);
          IF GETFILTER("Language Code") <> '' THEN
            IF Language.GET(ShopFilter,GETFILTER("Language Code")) THEN
              LanguageFilter := Language.Code;
          IF (LanguageFilter = '') AND (Shop."Default Language Code" <> '') THEN
            IF Language.GET(ShopFilter,Shop."Default Language Code") THEN
              LanguageFilter := Language.Code;
          IF LanguageFilter = '' THEN
            IF Language.FINDFIRST THEN
              LanguageFilter := Language.Code;
        END;

        // Filter f�r Shop und Sprache setzten
        IF LanguageFilter <> '' THEN BEGIN
          FILTERGROUP(4);
          SETRANGE("Shop Code",ShopFilter);
          SETRANGE("Language Code",LanguageFilter);
          FILTERGROUP(0);

          // Datensatz anlegen falls nicht vorhanden
          IF "Item No." <> '' THEN BEGIN
            ItemNo := "Item No."
          END ELSE BEGIN
            IF GETFILTER("Item No.") <> '' THEN
              ItemNo := GETFILTER("Item No.");
          END;
          IF ItemNo <> '' THEN BEGIN
            IF NOT GET(ShopFilter,LanguageFilter,ItemNo) THEN BEGIN
              INIT;
              VALIDATE("Shop Code",ShopFilter);
              VALIDATE("Language Code",LanguageFilter);
              VALIDATE("Item No.",ItemNo);
              INSERT(TRUE);
            END;
          END;
          CurrPage.UPDATE(FALSE);
        END;
      END;
    END;
