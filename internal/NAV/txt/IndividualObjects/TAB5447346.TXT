    PROCEDURE GetName@1000000000() Name : Text[80];
    VAR
      Setup@1000000006 : Record 5447300;
      CampaignHeader@1000000000 : Record 5447345;
      Shop@1000000004 : Record 5447301;
      ShopCode@1000000001 : Code[20];
      ItemShopCode@1000000002 : Code[20];
      CategoryShopCode@1000000003 : Code[20];
      LanguageCode@1000000005 : Code[10];
      Item@1000000007 : Record 5447304;
      Category@1000000008 : Record 5447303;
      ItemVariant@1000000009 : Record 5401;
    BEGIN
      Name := '';
      Setup.GET;
      CampaignHeader.GET("Link-to Header Code");

      IF CampaignHeader."Shop Code" <> '' THEN
        Shop.GET(CampaignHeader."Shop Code")
      ELSE
        Shop.GET(Setup."Default Shop Code");

      ShopCode := Shop.Code;
      ItemShopCode := Shop.Code;
      CategoryShopCode := Shop.Code;

      IF Shop."Use Items from Shop Code" <> '' THEN
        ItemShopCode := Shop."Use Items from Shop Code";

      IF Shop."Use Categories from Shop Code" <> '' THEN
        CategoryShopCode := Shop."Use Categories from Shop Code";

      IF CampaignHeader."Language Code" <> '' THEN
        LanguageCode := CampaignHeader."Language Code"
      ELSE
        LanguageCode := Shop."Default Language Code";

      //Artikel
      IF (Type = Type::"0") AND Item.GET(ItemShopCode,LanguageCode,"Item No.") THEN BEGIN
        Name := Item.Description;
        IF ("Item Variant Code" <> '') AND (ItemVariant.GET("Item No.","Item Variant Code")) THEN
          Name += ' ' + ItemVariant.Description;
      //Kategorie
      END ELSE IF (Type = Type::"1") AND (Category.GET(CategoryShopCode,LanguageCode,"Category Line No.")) THEN BEGIN
        Name := Category.Name;
      END;
    END;

    BEGIN
    {
      --- dynamic commerce - dynamic commerce GmbH - www.dc-solution.de ---
    }
    END.
