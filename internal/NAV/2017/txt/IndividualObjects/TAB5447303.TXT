      WebshopItem.SETRANGE("Shop Code",FromCategory."Shop Code");
      WebshopItem.SETRANGE("Language Code",FromCategory."Language Code");
      WebshopItem.SETRANGE("Main Category Line No.",SwapFromLineNo);
      IF WebshopItem.FINDSET THEN REPEAT
        SwapFromWebshopItem.INIT;
        SwapFromWebshopItem.TRANSFERFIELDS(WebshopItem);
        SwapFromWebshopItem."Main Category Line No." := SwapToLineNo;
        SwapFromWebshopItem.INSERT(FALSE);
      UNTIL WebshopItem.NEXT = 0;

      WebshopItem.INIT;
      WebshopItem.SETRANGE("Shop Code",FromCategory."Shop Code");
      WebshopItem.SETRANGE("Language Code",FromCategory."Language Code");
      WebshopItem.SETRANGE("Main Category Line No.",SwapToLineNo);
      IF WebshopItem.FINDSET THEN REPEAT
        SwapToWebshopItem.INIT;
        SwapToWebshopItem.TRANSFERFIELDS(WebshopItem);
        SwapToWebshopItem."Main Category Line No." := SwapFromLineNo;
        SwapToWebshopItem.INSERT(FALSE);
      UNTIL WebshopItem.NEXT = 0;

      IF SwapFromWebshopItem.FINDSET THEN REPEAT
      IF WebshopItem.GET(SwapFromWebshopItem."Shop Code",SwapFromWebshopItem."Language Code",SwapFromWebshopItem."Item No.") THEN BEGIN
        WebshopItem."Main Category Line No." := SwapFromWebshopItem."Main Category Line No.";
        WebshopItem.MODIFY(TRUE);
      END;
      UNTIL SwapFromWebshopItem.NEXT = 0;

      IF SwapToWebshopItem.FINDSET THEN REPEAT
      IF WebshopItem.GET(SwapToWebshopItem."Shop Code",SwapToWebshopItem."Language Code",SwapToWebshopItem."Item No.") THEN BEGIN
        WebshopItem."Main Category Line No." := SwapToWebshopItem."Main Category Line No.";
        WebshopItem.MODIFY(TRUE);
      END;
      UNTIL SwapToWebshopItem.NEXT = 0;


      // Kateogorie-Attribute �ber temp. Tabellen verschieben
      AttributeLink.RESET;
      AttributeLink.SETRANGE(Type,AttributeLink.Type::Category);
      AttributeLink.SETRANGE("Shop Code",FromCategory."Shop Code");
      AttributeLink.SETRANGE("Language Code",FromCategory."Language Code");
      AttributeLink.SETRANGE("Category Line No.",SwapFromLineNo);
      IF AttributeLink.FINDSET THEN REPEAT
        SwapFromAttributeLink.INIT;
        SwapFromAttributeLink.TRANSFERFIELDS(AttributeLink);
        SwapFromAttributeLink."Category Line No." := SwapToLineNo;
        SwapFromAttributeLink.INSERT;
      UNTIL AttributeLink.NEXT = 0;

      AttributeLink.RESET;
      AttributeLink.SETRANGE(Type,AttributeLink.Type::Category);
      AttributeLink.SETRANGE("Shop Code",FromCategory."Shop Code");
      AttributeLink.SETRANGE("Language Code",FromCategory."Language Code");
      AttributeLink.SETRANGE("Category Line No.",SwapToLineNo);
      IF AttributeLink.FINDSET THEN REPEAT
        SwapToAttributeLink.INIT;
        SwapToAttributeLink.TRANSFERFIELDS(AttributeLink);
        SwapToAttributeLink."Category Line No." := SwapFromLineNo;
        SwapToAttributeLink.INSERT;
      UNTIL AttributeLink.NEXT = 0;

      AttributeLink.RESET;
      AttributeLink.SETRANGE(Type,AttributeLink.Type::Category);
      AttributeLink.SETRANGE("Shop Code",FromCategory."Shop Code");
      AttributeLink.SETRANGE("Language Code",FromCategory."Language Code");
      AttributeLink.SETFILTER("Category Line No.",'%1|%2',SwapFromLineNo,SwapToLineNo);
      AttributeLink.DELETEALL(FALSE);

      AttributeLink.RESET;
      IF SwapFromAttributeLink.FINDSET THEN REPEAT
        AttributeLink.INIT;
        AttributeLink.TRANSFERFIELDS(SwapFromAttributeLink);
        AttributeLink.INSERT;
      UNTIL SwapFromAttributeLink.NEXT = 0;

      AttributeLink.RESET;
      IF SwapToAttributeLink.FINDSET THEN REPEAT
        AttributeLink.INIT;
        AttributeLink.TRANSFERFIELDS(SwapToAttributeLink);
        AttributeLink.INSERT;
      UNTIL SwapToAttributeLink.NEXT = 0;

      COMMIT;
    END;
