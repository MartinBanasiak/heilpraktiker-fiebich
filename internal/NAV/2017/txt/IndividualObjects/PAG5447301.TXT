          IF NoSeries.GET("Code No. Series Quotes") THEN BEGIN
            IF NoSeries."Manual Nos." THEN

          NoSeries.SETRANGE(NoSeries.Code,SalesSetup."Quote Nos.");
          IF NoSeries.FINDSET(FALSE,FALSE) THEN BEGIN
            IF NoSeries."Manual Nos." THEN
              UseOnlineOrderNoPossible := TRUE
            ELSE
              UseOnlineOrderNoError := STRSUBSTNO(Text003,FORMAT(SalesSetup."Quote Nos."));
          END ELSE
            UseOnlineOrderNoError := Text004;
        END;
      END ELSE BEGIN
        IF "Code No. Series Orders" <> '' THEN BEGIN
          IF NoSeries.GET("Code No. Series Orders") THEN BEGIN
            IF NoSeries."Manual Nos." THEN
              UseOnlineOrderNoPossible := TRUE
            ELSE
              UseOnlineOrderNoError := STRSUBSTNO(Text001,FORMAT("Code No. Series Orders"));
          END ELSE
            UseOnlineOrderNoError := STRSUBSTNO(Text002,FORMAT("Code No. Series Orders"));
        END ELSE BEGIN
          NoSeries.SETRANGE(NoSeries.Code,SalesSetup."Order Nos.");
          IF NoSeries.FINDSET(FALSE,FALSE) THEN BEGIN
            IF NoSeries."Manual Nos." THEN
              UseOnlineOrderNoPossible := TRUE
            ELSE
              UseOnlineOrderNoError := STRSUBSTNO(Text003,FORMAT(SalesSetup."Order Nos."));
          END ELSE
            UseOnlineOrderNoError := Text004;
        END;
      END;

      IF (NOT (UseOnlineOrderNoPossible)) AND ("Use Online Order No.") THEN
        ERROR(UseOnlineOrderNoError);
    END;

    BEGIN
    {
      --- dynamic commerce - dynamic commerce GmbH - www.dc-solution.de ---
    }
    END.
  }
