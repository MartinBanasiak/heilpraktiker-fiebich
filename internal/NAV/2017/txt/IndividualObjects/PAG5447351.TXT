    PROCEDURE GetCountWebshopItemsNoImage@1000000001() Counter : Integer;
    BEGIN
      Counter := 0;
      WebshopItem.RESET;
      //FILTERGROUP(4);
      WebshopItem.SETRANGE("Shop Code");
      WebshopItem.SETRANGE("Language Code");
      //FILTERGROUP(0);
      IF WebshopItem.FINDSET THEN REPEAT
        WebshopItem.CALCFIELDS("No. of Pictures","Is Variant");
        IF (NOT WebshopItem."Is Variant") AND (WebshopItem."No. of Pictures" = 0) THEN
          Counter += 1;
      UNTIL WebshopItem.NEXT = 0;
    END;

    PROCEDURE SetAvgTimeToShip@1000000000();
    VAR
      SalesShipmentHeader@1000000002 : Record 110;
      HeaderCounter@1000000000 : Integer;
      SumDays@1000000001 : Integer;
    BEGIN
      SalesShipmentHeader.RESET;
      //SalesShipmentHeader.SETFILTER("Webshop Shop Code",'<>%1','');
      SalesShipmentHeader.SETFILTER("Posting Date",'<>%1',0D);
      SalesShipmentHeader.SETFILTER("Order Date",'<>%1',0D);
      IF SalesShipmentHeader.FINDSET THEN REPEAT
        HeaderCounter += 1;
        SumDays += (SalesShipmentHeader."Posting Date" - SalesShipmentHeader."Order Date");
      UNTIL SalesShipmentHeader.NEXT = 0;
      IF HeaderCounter > 0 THEN
        AvgTimeToShip := ROUND((SumDays / HeaderCounter),0.01,'=')
      ELSE
        AvgTimeToShip := 0;
    END;
