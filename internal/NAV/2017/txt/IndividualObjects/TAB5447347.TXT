    LOCAL PROCEDURE UnapplyCampaignAdjustmentLineForSalesOrder@1000000008(AdjustmentLine@1000000000 : Record 5447347;SalesOrder@1000000001 : Record 36);
    VAR
      SalesLine@1000000002 : Record 37;
      Error001@1000000003 : TextConst 'DEU=Der �bergebene Sales Header Record muss dem in der Adjustment Zeile angegebenen eindeutigen Verkaufsauftrag entsprechen!;ENU=The passed Sales Header Record must match the unique Sales Order specified in the Adjustment Line!';
      SalesLineTmp@1000000011 : TEMPORARY Record 37;
      SignFactor@1000000004 : Integer;
      RevQtyAdjustment@1000000005 : Decimal;
      NewQty@1000000009 : Decimal;
      RevLineAmntAdjustment@1000000006 : Decimal;
      RevLineDiscPercentAdjustment@1000000007 : Decimal;
      RevUnitPriceAdjustment@1000000008 : Decimal;
      Deleted@1000000010 : Boolean;
    BEGIN
      IF (SalesOrder."Document Type" = SalesOrder."Document Type"::Order) AND
        SalesOrder.GET(SalesOrder."Document Type",SalesOrder."No.") AND
        (SalesOrder."No." = AdjustmentLine."Order No.") AND
        SalesLine.GET(SalesOrder."Document Type",SalesOrder."No.",AdjustmentLine."Line No.") THEN BEGIN
          SalesLineTmp := SalesLine;
          Deleted := FALSE;
          SalesLine.LOCKTABLE;

          IF AdjustmentLine."Previous Qty." <> AdjustmentLine."Adjusted Qty." THEN BEGIN
            RevQtyAdjustment := AdjustmentLine."Previous Qty." - AdjustmentLine."Adjusted Qty.";
            NewQty := SalesLine.Quantity + RevQtyAdjustment;
            IF (NewQty <= 0) THEN BEGIN
              SalesLine.DELETE(TRUE);
              Deleted := TRUE;
            END ELSE
              SalesLine.VALIDATE(Quantity,SalesLineTmp.Quantity + RevQtyAdjustment);
          END;

          IF (NOT Deleted) AND (AdjustmentLine."Previous Line Disc. %" <> AdjustmentLine."Adjusted Line Disc. %") THEN BEGIN
            RevLineDiscPercentAdjustment := AdjustmentLine."Previous Line Disc. %" - AdjustmentLine."Adjusted Line Disc. %";
            SalesLine.VALIDATE("Line Discount %",SalesLineTmp."Line Discount %" + RevLineDiscPercentAdjustment);
          END;

          IF (NOT Deleted) AND (AdjustmentLine."Previous Unit Price" <> AdjustmentLine."Adjusted Unit Price") THEN BEGIN
            RevUnitPriceAdjustment := AdjustmentLine."Previous Unit Price" - AdjustmentLine."Adjusted Unit Price";
            SalesLine.VALIDATE("Unit Price",SalesLineTmp."Unit Price" + RevUnitPriceAdjustment);
          END;

          IF (NOT Deleted) AND (AdjustmentLine."Previous Line Amnt." <> AdjustmentLine."Adjusted Line Amnt.") THEN BEGIN
            RevLineAmntAdjustment := AdjustmentLine."Previous Line Amnt." - AdjustmentLine."Adjusted Line Amnt.";
            SalesLine.VALIDATE("Line Amount",SalesLineTmp."Line Amount" + RevLineAmntAdjustment);
          END;

          IF (NOT Deleted) THEN
            SalesLine.MODIFY(TRUE);
      END ELSE
       ERROR(Error001);
    END;
