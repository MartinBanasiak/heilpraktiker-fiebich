                                  WebshopManagement.StateIndicatorUpdate();

                                    FileType::Document: Filename := 'pdf';
                                    FileType::Videoclip: Filename := 'flv';
                                  END;

                                  TempFolder.SETRANGE(Path,FilePath);
                                  TempFolder.SETRANGE("Is a file",TRUE);
                                  Filename := STRSUBSTNO('%1*%2',Item."No.",Filename);
                                  TempFolder.SETFILTER(Name,Filename);
                                  IF TempFolder.FINDSET THEN REPEAT
                                    IF FILE.EXISTS(FilePath + TempFolder.Name) THEN BEGIN
                                      IF InsertFiles THEN BEGIN
                                        // Webshop Artikel anlegen falls n�tig
                                        IF NOT WebshopItem.GET(ShopCode,LanguageCode,"No.") THEN BEGIN
                                          WebshopItem.INIT;
                                          WebshopItem.VALIDATE("Shop Code",ShopCode);
                                          WebshopItem.VALIDATE("Language Code",LanguageCode);
                                          WebshopItem.VALIDATE("Item No.","No.");
                                          WebshopItem.INSERT(TRUE);
                                        END;
                                        // Webshop Artikel aktiv setzten (Optional)
                                        IF ActivateItem THEN BEGIN
                                          WebshopItem.VALIDATE("Is Active",TRUE);
                                          WebshopItem.MODIFY(TRUE);
                                        END;
                                        ItemNo := WebshopItem."Item No.";
                                        ItemDescription := WebshopItem.Description;

                                        // Datei anlegen
                                        WebshopItemFile.SETRANGE(Type,FileType);
                                        WebshopItemFile.SETRANGE("Shop Code",WebshopItem."Shop Code");
                                        WebshopItemFile.SETRANGE("Language Code",WebshopItem."Language Code");
                                        WebshopItemFile.SETRANGE("Item No.",WebshopItem."Item No.");
                                        WebshopItemFile.SETFILTER("Temp Filename",'%1',TempFolder.Name);
                                        IF NOT WebshopItemFile.FINDSET THEN BEGIN
                                          WebshopItemFile.SETRANGE("Temp Filename");
                                          IF WebshopItemFile.FINDLAST THEN BEGIN
                                            NextLineNo := WebshopItemFile."Line No." + 10000
                                          END ELSE BEGIN
                                            NextLineNo := 10000;
                                          END;
                                          NewWebshopItemFile.INIT;
                                          NewWebshopItemFile."Shop Code" := WebshopItem."Shop Code";
                                          NewWebshopItemFile."Language Code" := WebshopItem."Language Code";
                                          NewWebshopItemFile."Item No." := WebshopItem."Item No.";
                                          NewWebshopItemFile.Type := FileType;
                                          NewWebshopItemFile."Line No." := NextLineNo;
                                          NewWebshopItemFile.INSERT(TRUE);
                                          NewWebshopItemFile.Description := WebshopItem.Description;
                                          NewWebshopItemFile."All Language Codes" := AllLanguageCodes;
                                          NewWebshopItemFile."Temp Filename" := TempFolder.Name;
                                          NewWebshopItemFile.MODIFY(TRUE);
                                          InsertCounter += 1;
                                        END;

                                        // Artikel aktualisieren (Optional)
                                        IF UpdateItem THEN BEGIN
                                          WebshopUpdateManagement.UpdateWebshopItem(WebshopItem);
                                          WebshopUpdateManagement.UpdateWebshopItemFile(NewWebshopItemFile);
                                          WebshopUpdateManagement.CheckWebshopItemFile(NewWebshopItemFile);
                                        END;
                                      END;
                                      Counter += 1;
                                    END;

                                  UNTIL TempFolder.NEXT = 0;
                                END;
