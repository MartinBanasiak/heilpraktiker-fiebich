                                    END;
                                  END;

                                  IF FromAttributeLink.FINDSET THEN BEGIN
                                    REPEAT

                                                                 FromAttributeLink."Attribute Code",FromAttributeLink."Value Option") THEN BEGIN
                                        ToAttributeLink.INIT;
                                        ToAttributeLink.TRANSFERFIELDS(FromAttributeLink);
                                        ToAttributeLink.Type := ToAttributeLink.Type::Item;
                                        ToAttributeLink."No." := "Item No.";
                                        ToAttributeLink.Sorting := FromAttributeLink.Sorting;
                                        CASE FromType OF
                                           FromType::Template: BEGIN
                                              ToAttributeLink."Shop Code" := "Shop Code";
                                              ToAttributeLink."Language Code" := "Language Code";
                                           END;
                                          FromType::Item: BEGIN
                                              ToAttributeLink."Shop Code" := FromAttributeLink."Shop Code";
                                              ToAttributeLink."Language Code" := FromAttributeLink."Language Code";
                                          END;
                                        END;
                                        ToAttributeLink.INSERT(TRUE);
                                        Counter += 1;
                                      END;
                                    UNTIL FromAttributeLink.NEXT = 0;
                                    IF FromAttributeTranslation.FINDSET THEN REPEAT
                                      IF NOT ToAttributeTranslation.GET(ToAttributeTranslation.Type::"Attribute Link",FromAttributeTranslation."Attribute Code",
                                                                        "Item No.",FromAttributeTranslation."Language Code") THEN BEGIN
                                        ToAttributeTranslation.INIT;
                                        ToAttributeTranslation.TRANSFERFIELDS(FromAttributeTranslation);
                                        ToAttributeTranslation.Type := ToAttributeTranslation.Type::"Attribute Link";
                                        ToAttributeTranslation."Attribute Link No." := "Item No.";
                                        ToAttributeTranslation.INSERT(TRUE);
                                      END;
                                    UNTIL FromAttributeTranslation.NEXT = 0;
                                  END;
                                END;
