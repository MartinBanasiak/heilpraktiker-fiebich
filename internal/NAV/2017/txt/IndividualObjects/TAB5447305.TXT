    PROCEDURE GetItemCategoryPath@1000000002() ItemCategoryURL : Text[1024];
    VAR
      Category@1000000000 : Record 5447303;
      WebshopItem@1000000001 : Record 5447304;
      CategoryURL@1000000003 : Text[512];
      ItemURL@1000000002 : Text[512];
    BEGIN
      IF Category.GET("Category Shop Code","Category Language Code","Category Line No.") THEN
        CategoryURL := Category.GetCategoryURL;

    PROCEDURE SetItemCategoriesWithAncestors@1000000003(CategoryShopCode@1000000000 : Code[20];LanguageCode@1000000001 : Code[10];ItemNo@1000000002 : Code[20];VAR ItemCategoriesWithAncestors@1000000003 : TEMPORARY Record 5447303);
    VAR
      WebshopCategory@1000000004 : Record 5447303;
    BEGIN
      SETRANGE("Category Shop Code",CategoryShopCode);
      SETRANGE("Category Language Code",LanguageCode);
      SETRANGE("Item No.",ItemNo);
      IF FINDSET(FALSE,FALSE) THEN REPEAT
        IF WebshopCategory.GET(CategoryShopCode,LanguageCode,"Category Line No.") AND NOT (ItemCategoriesWithAncestors.GET(CategoryShopCode,LanguageCode,"Category Line No.")) THEN BEGIN
          ItemCategoriesWithAncestors.INIT;
          ItemCategoriesWithAncestors := WebshopCategory;
          ItemCategoriesWithAncestors.INSERT(FALSE);
          WebshopCategory.SetAncestors(CategoryShopCode,LanguageCode,"Category Line No.",ItemCategoriesWithAncestors);
        END;
      UNTIL NEXT = 0;
    END;
